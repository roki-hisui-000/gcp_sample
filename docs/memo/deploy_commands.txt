----------------------------------------------------------------------------------------------------
- GCP デプロイ 手順（2コンテナ分割構成版）
----------------------------------------------------------------------------------------------------

#-------------------------------------------------
# 前提
#-------------------------------------------------

プロジェクトID：gcp-sample-project-509110
ネットワークポリシー：my-valkey-policy
valkeyインスタンス名：my-valkey
リージョン：asia-northeast1

#-------------------------------------------------
# 1. ログイン & APIの有効化
#-------------------------------------------------

# ログイン
gcloud auth login

# ログイン確認
gcloud auth print-access-token | docker login -u oauth2accesstoken --password-stdin https://asia-northeast1-docker.pkg.dev

# 必要なネットワークAPIの有効化
gcloud services enable \
    networkconnectivity.googleapis.com \
    serviceconsumermanagement.googleapis.com \
    vpcaccess.googleapis.com \
    run.googleapis.com \
    artifactregistry.googleapis.com \
    cloudbuild.googleapis.com

#-------------------------------------------------
# 2. VPC接続とValkey（Memorystore）の準備
#-------------------------------------------------

# サービス接続ポリシーの作成（既に存在する場合はエラーが出ますがスキップして問題ありません）
gcloud network-connectivity service-connection-policies create my-valkey-policy \
    --network=default \
    --region=asia-northeast1 \
    --service-class=gcp-memorystore \
    --subnets=default

# Valkeyインスタンスを作成（数分かかる、既に存在する場合はスキップ）
gcloud memorystore instances create my-valkey \
    --location=asia-northeast1 \
    --node-type=shared-core-nano \
    --shard-count=1 \
    --replica-count=0 \
    --mode=cluster-disabled \
    --endpoints='[{"connections": [{"pscAutoConnection": {"network": "projects/gcp-sample-project-509110/global/networks/default", "projectId": "gcp-sample-project-509110"}}]}]'

# ⚠️ 非常に重要 ⚠️
# インスタンスの詳細情報を取得し、必ず「PRIMARY」ノード의 IPアドレスを控えてください。
# 「READER」ノードのIPは読み取り専用のため、セッションの書き込み（ログイン）時にエラー500になります。
gcloud memorystore instances describe my-valkey --location=asia-northeast1

# 出力例（"CONNECTION_TYPE_PRIMARY" の下にある ipAddress を確認すること）：
# connections:
# - pscAutoConnection:
#     connectionType: CONNECTION_TYPE_PRIMARY
#     ipAddress: 10.146.0.10  <-- これを控える

#-------------------------------------------------
# 3. Dockerリポジトリの作成（初回のみ）
#-------------------------------------------------

# "Artifact Registry"にリポジトリを新規作成する
gcloud artifacts repositories create gcp-sample-repo \
    --repository-format=docker \
    --location=asia-northeast1 \
    --description="Docker repository for gcp-sample"

#-------------------------------------------------
# 4. Backend (APIサーバー) のビルド・PUSH・デプロイ
#-------------------------------------------------

# ビルド（本番環境向け linux/amd64）
docker build --platform linux/amd64 -t asia-northeast1-docker.pkg.dev/gcp-sample-project-509110/gcp-sample-repo/backend:latest ./backend

# PUSH
docker push asia-northeast1-docker.pkg.dev/gcp-sample-project-509110/gcp-sample-repo/backend:latest

# Cloud Runへデプロイ（ValkeyのPRIMARYノードのIPを指定） ⭐️VALKEY_HOSTに指定するIPアドレスは先ほど取得したものを指定⭐️
# デプロイ後に表示される "Service URL" をコピーして控えてください（Frontendのデプロイで使います）。
gcloud run deploy gcp-sample-backend \
    --image asia-northeast1-docker.pkg.dev/gcp-sample-project-509110/gcp-sample-repo/backend:latest \
    --region asia-northeast1 \
    --network=default \
    --subnet=default \
    --set-env-vars VALKEY_HOST=10.146.0.15,VALKEY_PORT=6379,SESSION_SECRET=my-secure-session-secret-key-123 \
    --allow-unauthenticated

#-------------------------------------------------
# 5. Frontend (Gateway/プロキシ兼静的配信) のビルド・PUSH・デプロイ
#-------------------------------------------------

# ビルド（本番環境向け linux/amd64）
docker build --platform linux/amd64 -t asia-northeast1-docker.pkg.dev/gcp-sample-project-509110/gcp-sample-repo/frontend:latest ./frontend

# PUSH
docker push asia-northeast1-docker.pkg.dev/gcp-sample-project-509110/gcp-sample-repo/frontend:latest

# Cloud Runへデプロイ（BACKEND_URLに上記でコピーしたバックエンドのサービスURLを指定）
# フロントエンドは直接データベースに触らないため、VPCネットワークの設定（--network, --subnet）は不要です。
gcloud run deploy gcp-sample-frontend \
    --image asia-northeast1-docker.pkg.dev/gcp-sample-project-509110/gcp-sample-repo/frontend:latest \
    --region asia-northeast1 \
    --set-env-vars BACKEND_URL=https://gcp-sample-backend-485401255876-an.a.run.app \
    --allow-unauthenticated

#-------------------------------------------------
# 6. 疎通・動作確認方法
#-------------------------------------------------
# デプロイしたフロントエンドのURLにブラウザからアクセスします。
# ログインが成功し、メッセージが送信されれば、2コンテナの連携は完璧です。

#----------------------------------------------------------------------------------------------------
# - サービス停止 / 削除
#----------------------------------------------------------------------------------------------------

# Cloud Run サービスの削除 (2つ)
gcloud run services delete gcp-sample-frontend --region=asia-northeast1 --quiet
gcloud run services delete gcp-sample-backend --region=asia-northeast1 --quiet

# Valkey インスタンスの削除（数分かかる）
gcloud memorystore instances delete my-valkey --location=asia-northeast1 --quiet

# ネットワーク接続ポリシーの削除
gcloud network-connectivity service-connection-policies delete my-valkey-policy \
    --region=asia-northeast1 \
    --quiet

# Artifact Registryにアップロードしたdockerイメージを削除
gcloud artifacts docker images delete \
    asia-northeast1-docker.pkg.dev/gcp-sample-project-509110/gcp-sample-repo/frontend:latest \
    --delete-tags \
    --quiet

gcloud artifacts docker images delete \
    asia-northeast1-docker.pkg.dev/gcp-sample-project-509110/gcp-sample-repo/backend:latest \
    --delete-tags \
    --quiet
